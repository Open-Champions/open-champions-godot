cmake_minimum_required(VERSION 3.21)
project(champions_godot_archive VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(GODOT_CPP_DIR "" CACHE PATH "Checkout of the godot-cpp commit in dependencies.json")
if(NOT EXISTS "${GODOT_CPP_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set GODOT_CPP_DIR to the pinned godot-cpp checkout (see dependencies.json).")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GODOT_CPP_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE GODOT_CPP_REV OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE PIN_RESULT)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.json" DEPENDENCY_LOCK)
string(JSON GODOT_CPP_PIN GET "${DEPENDENCY_LOCK}" godot_cpp_commit)
if(NOT PIN_RESULT EQUAL 0 OR NOT GODOT_CPP_REV STREQUAL GODOT_CPP_PIN)
    message(FATAL_ERROR "godot-cpp must be checked out at ${GODOT_CPP_PIN}")
endif()
# The archive API uses C++ exceptions; retain them across the adapter's catch boundary.
set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "Preserve archive error handling" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "Trim bindings" FORCE)
set(GODOTCPP_SYSTEM_HEADERS ON CACHE BOOL "Treat upstream bindings as system headers" FORCE)
set(GODOTCPP_PRECISION "single" CACHE STRING "Pinned Godot standard build precision" FORCE)
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "Restart after rebuilding the extension" FORCE)
# Desktop packages use a single static release CRT across the binding and core.
# Pin these before target creation so a reused cache cannot switch to /MD.
if(MSVC)
    set(GODOTCPP_USE_STATIC_CPP ON CACHE BOOL "Self-contained desktop CRT" FORCE)
    set(GODOTCPP_DEBUG_CRT OFF CACHE BOOL "Use release CRT consistently" FORCE)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded" CACHE STRING "Static release CRT" FORCE)
endif()
add_subdirectory("${GODOT_CPP_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/godot-cpp")
# Reuse the existing implementation. The standalone native build remains Godot-free.
add_library(champions-archive-binding-core STATIC ../native/src/archive.cpp)
target_include_directories(champions-archive-binding-core PUBLIC ../native/include)
set_target_properties(champions-archive-binding-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(champions-archive-extension SHARED src/archive_inspector.cpp)
target_link_libraries(champions-archive-extension PRIVATE champions-archive-binding-core godot::cpp)
if(MSVC)
    target_compile_options(champions-archive-extension PRIVATE /W4 /utf-8 /EHsc)
    target_compile_options(champions-archive-binding-core PRIVATE /W4 /utf-8 /EHsc)
else()
    target_compile_options(champions-archive-extension PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(champions-archive-binding-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
# Use the configured target: godot_arch_name relies on godot-cpp directory scope.
get_target_property(CHAMPIONS_ARCH godot-cpp GODOTCPP_ARCH)
if(NOT CHAMPIONS_ARCH)
    message(FATAL_ERROR "godot-cpp did not expose its configured architecture")
endif()
if(APPLE)
    set(CHAMPIONS_PLATFORM "macos")
    # Official macOS export templates are Universal 2. Export builds must supply
    # CMAKE_OSX_ARCHITECTURES=arm64;x86_64; host-only builds retain this filename.
    set(CHAMPIONS_LIBRARY_NAME "champions_archive.macos")
elseif(WIN32)
    set(CHAMPIONS_PLATFORM "windows")
else()
    set(CHAMPIONS_PLATFORM "linux")
endif()
if(NOT APPLE)
    set(CHAMPIONS_LIBRARY_NAME "champions_archive.${CHAMPIONS_PLATFORM}.${CHAMPIONS_ARCH}")
endif()
set_target_properties(champions-archive-extension PROPERTIES
    PREFIX ""
    OUTPUT_NAME "${CHAMPIONS_LIBRARY_NAME}"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../godot/bin/$<0:>"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../godot/bin/$<0:>")
