cmake_minimum_required(VERSION 3.21)
project(champions_godot_archive VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# CMAKE_BUILD_TYPE controls optimization, not Godot's allocator/build ABI.
# Editor and debug templates share template_debug; release exports require
# template_release. Separate build directories produce both named libraries.
set(GODOTCPP_TARGET "template_debug" CACHE STRING "Godot host: template_debug or template_release")
set_property(CACHE GODOTCPP_TARGET PROPERTY STRINGS "template_debug;template_release")
if(NOT GODOTCPP_TARGET STREQUAL "template_debug" AND NOT GODOTCPP_TARGET STREQUAL "template_release")
    message(FATAL_ERROR "GODOTCPP_TARGET must be template_debug or template_release")
endif()
set(GODOT_CPP_DIR "" CACHE PATH "Checkout of the godot-cpp commit in dependencies.json")
if(NOT EXISTS "${GODOT_CPP_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set GODOT_CPP_DIR to the pinned godot-cpp checkout (see dependencies.json).")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GODOT_CPP_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE GODOT_CPP_REV OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE PIN_RESULT)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.json" DEPENDENCY_LOCK)
string(JSON GODOT_CPP_PIN GET "${DEPENDENCY_LOCK}" godot_cpp_commit)
if(NOT PIN_RESULT EQUAL 0 OR NOT GODOT_CPP_REV STREQUAL GODOT_CPP_PIN)
    message(FATAL_ERROR "godot-cpp must be checked out at ${GODOT_CPP_PIN}")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/source_path_map.cmake")
# The archive API uses C++ exceptions; retain them across the adapter's catch boundary.
set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "Preserve archive error handling" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "Trim bindings" FORCE)
set(GODOTCPP_SYSTEM_HEADERS ON CACHE BOOL "Treat upstream bindings as system headers" FORCE)
set(GODOTCPP_PRECISION "single" CACHE STRING "Pinned Godot standard build precision" FORCE)
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "Restart after rebuilding the extension" FORCE)
# Desktop packages use a single static release CRT across the binding and core.
# Pin these before target creation so a reused cache cannot switch to /MD.
if(MSVC)
    set(GODOTCPP_USE_STATIC_CPP ON CACHE BOOL "Self-contained desktop CRT" FORCE)
    set(GODOTCPP_DEBUG_CRT OFF CACHE BOOL "Use release CRT consistently" FORCE)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded" CACHE STRING "Static release CRT" FORCE)
endif()
add_subdirectory("${GODOT_CPP_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/godot-cpp")
# Reuse the existing implementation. The standalone native build remains Godot-free.
add_library(champions-archive-binding-core STATIC ../native/src/ui_font.cpp ../native/src/ui_art_shape.cpp ../native/src/ui_font_page.cpp ../native/src/ui_layout.cpp ../native/src/ui_text_layout.cpp ../native/src/hud_dynamic.cpp ../native/src/hud_round3.cpp ../native/src/hud_extras.cpp ../native/src/ui_composite.cpp ../native/src/archive.cpp ../native/src/resource_manager.cpp ../native/src/texture.cpp ../native/src/texture_upscale.cpp ../native/src/inventory_icons.cpp ../native/src/inventory_icon_assoc.cpp ../native/src/model.cpp ../native/src/animation.cpp ../native/src/skinning.cpp ../native/src/world_geometry.cpp)
target_include_directories(champions-archive-binding-core PUBLIC ../native/include)
set_target_properties(champions-archive-binding-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
# Link exactly the same deterministic target graph used by native tests.
set(CHAMPIONS_ACTORS_ARCHIVE champions-archive-binding-core)
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/simulation_targets.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/player_state_targets.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/level_flow_targets.cmake")
add_library(champions-native-fx ../native/src/fx/fx.cpp ../native/src/fx/fx_json.cpp ../native/src/lighting/lighting.cpp ../native/src/lighting/level_lights.cpp)
target_include_directories(champions-native-fx PUBLIC ../native/include)
set_target_properties(champions-native-fx PROPERTIES POSITION_INDEPENDENT_CODE ON)
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/fx2_targets.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/script_commands_targets.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/campaign_fight_targets.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/combat_play_effects_targets.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/fixture_save_targets.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/ps2_save_targets.cmake")
foreach(target champions-fx-exact champions-fx-render champions-fx-recipes champions-fx-skills champions-fx-sim champions-combat-play-effects)
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
foreach(target IN LISTS CHAMPIONS_SIMULATION_TARGETS)
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/menus_targets.cmake")
add_library(champions-native-audio-binding-core STATIC ../native/src/audio/native_audio.cpp ../native/src/audio/music_volume.cpp)
target_include_directories(champions-native-audio-binding-core PUBLIC ../native/include)
set_target_properties(champions-native-audio-binding-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
include("${CMAKE_CURRENT_SOURCE_DIR}/../native/frontend_targets.cmake")
add_library(champions-archive-extension SHARED src/archive_inspector.cpp src/session_controller.cpp src/session_controller_text.cpp src/ui_layout_service.cpp src/ui_layout_dynamic_service.cpp src/ui_hud_round3_service.cpp src/hud_extras_service.cpp src/menu_service.cpp src/frontend_service.cpp src/fx_bridge.cpp src/native_audio_bridge.cpp src/level_lights_service.cpp src/ps2_save_import.cpp)
target_sources(champions-archive-extension PRIVATE ../native/src/real_world.cpp)
target_sources(champions-archive-extension PRIVATE src/session_controller_hub.cpp)
target_link_libraries(champions-archive-extension PRIVATE champions-player-state champions-menus champions-frontend champions-play-fixture-save champions-fx-world-save champions-level-play champions-combat-play champions-player-services champions-archive-binding-core champions-native-save champions-terrain-file champions-native-fx champions-fx-sim champions-script-commands champions-campaign-fight champions-combat-play-effects champions-native-audio-binding-core champions-ps2-save godot::cpp)
# Match the standalone animation target's authored binary64 numeric policy.
# Apply it across the bridge as well so finite-value checks remain meaningful.
if(MSVC)
    target_compile_options(champions-native-audio-binding-core PRIVATE /W4 /fp:strict)
    target_compile_options(champions-native-fx PRIVATE /W4 /utf-8 /fp:strict)
    target_compile_options(champions-archive-extension PRIVATE /W4 /utf-8 /EHsc /fp:strict)
    target_compile_options(champions-archive-binding-core PRIVATE /W4 /utf-8 /EHsc /fp:strict)
else()
    target_compile_options(champions-native-audio-binding-core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
    target_compile_options(champions-native-fx PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
    target_compile_options(champions-archive-extension PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
    target_compile_options(champions-archive-binding-core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
# Use the configured target: godot_arch_name relies on godot-cpp directory scope.
get_target_property(CHAMPIONS_ARCH godot-cpp GODOTCPP_ARCH)
if(NOT CHAMPIONS_ARCH)
    message(FATAL_ERROR "godot-cpp did not expose its configured architecture")
endif()
if(APPLE)
    set(CHAMPIONS_PLATFORM "macos")
    # Official macOS export templates are Universal 2. Export builds must supply
    # CMAKE_OSX_ARCHITECTURES=arm64;x86_64; host-only builds retain this filename.
    set(CHAMPIONS_LIBRARY_NAME "champions_archive.macos.${GODOTCPP_TARGET}")
elseif(WIN32)
    set(CHAMPIONS_PLATFORM "windows")
else()
    set(CHAMPIONS_PLATFORM "linux")
endif()
if(NOT APPLE)
    set(CHAMPIONS_LIBRARY_NAME "champions_archive.${CHAMPIONS_PLATFORM}.${GODOTCPP_TARGET}.${CHAMPIONS_ARCH}")
endif()
set_target_properties(champions-archive-extension PROPERTIES
    PREFIX ""
    OUTPUT_NAME "${CHAMPIONS_LIBRARY_NAME}"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../godot/bin/$<0:>"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../godot/bin/$<0:>")
