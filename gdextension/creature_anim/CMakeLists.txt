cmake_minimum_required(VERSION 3.21)
project(champions_creature_animation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GODOT_CPP_DIR "" CACHE PATH "Pinned godot-cpp checkout")
set(GODOTCPP_TARGET "template_debug" CACHE STRING "Godot ABI")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GODOT_CPP_DIR}" rev-parse HEAD OUTPUT_VARIABLE REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies.json" LOCK)
string(JSON PIN GET "${LOCK}" godot_cpp_commit)
if(NOT REV STREQUAL PIN)
  message(FATAL_ERROR "Use the pinned godot-cpp checkout: ${PIN}")
endif()
set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "Exceptions cross the adapter" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/../build_profile.json" CACHE PATH "Trim bindings" FORCE)
set(GODOTCPP_SYSTEM_HEADERS ON CACHE BOOL "System headers" FORCE)
add_subdirectory("${GODOT_CPP_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/godot-cpp")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../native/simulation_targets.cmake")
foreach(t IN LISTS CHAMPIONS_SIMULATION_TARGETS)
  set_target_properties(${t} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
add_library(champions-creature-animation SHARED binding.cpp ../../native/src/model.cpp)
target_include_directories(champions-creature-animation PRIVATE ../../native/include)
target_link_libraries(champions-creature-animation PRIVATE champions-creature-anim godot::cpp)
if(MSVC)
  target_compile_options(champions-creature-animation PRIVATE /W4 /EHsc /fp:strict)
else()
  target_compile_options(champions-creature-animation PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
set_target_properties(champions-creature-animation PROPERTIES OUTPUT_NAME "champions-creature-animation.${GODOTCPP_TARGET}" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../godot/creature_anim/bin" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../godot/creature_anim/bin")
